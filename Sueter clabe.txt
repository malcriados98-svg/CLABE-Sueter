<script>
// REEMPLAZA ESTOS NÚMEROS CON TU CLABE REAL (18 DÍGITOS)
const miClabe = "012345678901234567";

function copiarClabe() {
navigator.clipboard.writeText(miClabe).then(() => {
document.getElementById("mensaje").innerText = "¡CLABE Copiada con éxito! ✅";
}).catch(err => {
document.getElementById("mensaje").innerText = "Error al copiar automáticamente ❌";
console.error("Error: ", err);
});
}

// Ejecutar copia automática al cargar la página
window.onload = function() {
// Pequeño retraso para asegurar que el navegador responda bien
setTimeout(copiarClabe, 500);
};
</script>
